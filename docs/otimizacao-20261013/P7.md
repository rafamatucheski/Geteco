## Interruptor e comportamento

`GETECO_SKID_BUDGET=0` desliga a otimização antes de iniciar o processo. Ausente ou diferente de `0`, liga por padrão. O valor é lido uma vez no carregamento do script.

Ligado: cada marca entra no anel compartilhado; quando cheio, sai primeiro o segmento inserido há mais tempo, inclusive entre carros diferentes. A lista `marks` de cada carro continua disponível, contendo apenas suas marcas ainda retidas. Expiração percorre até 256 entradas, respeitando borracha de 10 s e sulcos de 30 s; não assume que a ordem de inserção corresponde à ordem de vencimento.

Uma `ArrayMesh`, uma superfície, um `MeshInstance3D` sem sombra e um timer pausável de 0,1 s servem ao mundo principal. O anel fica na raiz da árvore; remoção/limpeza de um carro libera somente suas entradas por referência fraca. Quando vazio, libera superfícies, esconde a malha e para o timer; novo mundo reutiliza o material existente sem rastros antigos.

Redesenho acontece quando entram/saem segmentos ou quando o fade altera as cores RGBA8. Chamadas sem mudança não reenviam a geometria. Cantos, normais, largura, profundidade, deslocamento sobre a superfície e fórmulas de cor/fade preservam o legado. Continua usando exclusivamente `VehicleTireEffects.mark_material()`.

Desligado: limite original de 320 segmentos por carro, malha/timer por carro e redesenho original de 0,1 s, sem registrar entradas globais.

## Reprodução do contrato temporário

Copiar o bloco abaixo para um `.gd` temporário fora do repositório e usar o comando acima; não exige alterar arquivos pertencentes a outras tarefas.

```gdscript
extends SceneTree
const TIRES = preload("res://gameplay/vehicle_effects/VehicleTireEffects.gd")
const BUDGET = preload("res://gameplay/vehicle_effects/SkidmarkBudget.gd")
var checks := 0
var errors: Array[String] = []
class Car extends CharacterBody3D:
	var handling := {"mass": 1.0}
func _initialize() -> void: run.call_deferred()
func check(ok: bool, label: String) -> void:
	checks += 1
	if not ok: errors.append(label)
func run() -> void:
	var world := Node3D.new()
	root.add_child(world)
	var tires: Array[Node] = []
	for index in 10:
		var car := Car.new()
		world.add_child(car)
		var tire := TIRES.new()
		tire.configure(car)
		car.add_child(tire)
		tires.append(tire)
	# Contatos intercalados reproduzem dez carros deixando dois rastros.
	for step in 321:
		for index in 10:
			for side in 2:
				tires[index]._append_mark(side, {"point":Vector3(index*4+side,0,step*.1),"normal":Vector3.UP}, "hard")
	var total := 0
	for tire in tires: total += tire.marks.size()
	check(total == (256 if BUDGET.enabled else 3200), "contagem mundial esperada")
	for tire in tires: tire._redraw_marks()
	var meshes := root.find_children("*TireMarks", "MeshInstance3D", true, false)
	var timers := root.find_children("*", "Timer", true, false)
	var mats := {}
	for mesh in meshes: mats[mesh.material_override.get_instance_id()] = true
	check(meshes.size() == (1 if BUDGET.enabled else 10), "malhas esperadas")
	check(mats.size() == 1, "material único")
	if BUDGET.enabled:
		var budget = BUDGET._instance
		check(budget.segment_count == 256, "anel global limitado")
		check(tires[0].marks.front().a.z > 30, "mais antigos descartados primeiro")
		check(budget.geometry.surface_get_arrays(0)[Mesh.ARRAY_VERTEX].size() == 1536, "seis vértices por segmento")
		# Vida longa mantém RGBA8 estável durante chamadas consecutivas.
		for tire in tires:
			for mark in tire.marks: mark.life = 1000000000
		budget._redraw()
		var stable = budget.redraw_count
		for iteration in 10: budget._redraw()
		check(budget.redraw_count == stable, "sem upload quando nada mudou")
		var before = budget.segment_count
		tires[0].clear_all()
		check(budget.segment_count < before and budget.segment_count > 0, "limpeza de um carro preserva os outros")
		budget._redraw()
		for tire in tires: tire.clear_all()
		check(budget.segment_count == 0 and budget._timer.is_stopped() and budget.geometry.get_surface_count() == 0, "vazio libera superfícies e para timer")
		# Vida mista: sulco mais velho não impede expiração da borracha nova.
		for index in 3: tires[0]._append_mark(0, {"point":Vector3(0,0,index*.1),"normal":Vector3.UP}, "grass", true)
		for index in 3: tires[1]._append_mark(0, {"point":Vector3(4,0,index*.1),"normal":Vector3.UP}, "hard")
		for mark in tires[0].marks: mark.born = Time.get_ticks_msec()-11000
		for mark in tires[1].marks: mark.born = Time.get_ticks_msec()-11000
		budget._tick()
		check(tires[0].marks.size() == 2 and tires[1].marks.is_empty(), "idade remove borracha no meio e mantém sulcos")
		var color = budget.geometry.surface_get_arrays(0)[Mesh.ARRAY_COLOR][0]
		for mark in tires[0].marks: mark.born -= 5000
		budget._tick()
		check(budget.geometry.surface_get_arrays(0)[Mesh.ARRAY_COLOR][0].a < color.a, "fade continua por idade sem novos contatos")
		for mark in tires[0].marks: mark.born = Time.get_ticks_msec()-31000
		budget._tick()
		check(budget.segment_count == 0 and not budget.mark_mesh.visible, "sulcos também expiram")
		for index in 3: tires[2]._append_mark(0, {"point":Vector3(8,0,index*.1),"normal":Vector3.UP}, "hard")
		budget._redraw()
		world.queue_free()
		await process_frame
		await process_frame
		check(budget.segment_count == 0 and budget._timer.is_stopped(), "remoção do mundo limpa donos e timer")
		var next_world := Node3D.new()
		root.add_child(next_world)
		var next_car := Car.new()
		next_world.add_child(next_car)
		var next_tire := TIRES.new()
		next_tire.configure(next_car)
		next_car.add_child(next_tire)
		for index in 3: next_tire._append_mark(0, {"point":Vector3(0,0,index*.1),"normal":Vector3.UP}, "hard")
		budget._redraw()
		check(budget.segment_count == 2 and not budget._timer.is_stopped() and budget.mark_mesh.material_override == TIRES.mark_material(), "nova sessão reutiliza material sem marcas antigas")
		next_world.queue_free()
		await process_frame
		await process_frame
		check(budget.segment_count == 0, "segunda sessão também limpa o orçamento")
	else:
		world.queue_free()
		await process_frame
		await process_frame
	print("P7_CONTRACT ", JSON.stringify({"enabled":BUDGET.enabled,"cars":10,"segments":total,"meshes":meshes.size(),"timers":timers.size(),"materials":mats.size(),"checks":checks,"errors":errors}))
	quit(0 if errors.is_empty() else 1)
```
