extends CanvasLayer
## Presentation adapter for the production HUD.
##
## FullSession and Driving retain ownership of their existing Label references.
## This layer reads those compatibility nodes and the same authoritative state,
## then presents each channel in a stable, responsive region.

const WEAPONS := preload("res://gameplay/WeaponCatalog.gd")
const FONT_REGULAR := preload("res://assets/fonts/barlow/BarlowSemiCondensed-Regular.ttf")
const FONT_SEMIBOLD := preload("res://assets/fonts/barlow/BarlowSemiCondensed-SemiBold.ttf")

const TEXT := Color("f3f0e7")
const MUTED := Color("b9c2c8")
const MONEY := Color("75c97a")
const WANTED := Color("f6c445")
const HEALTH := Color("c83e42")
const ARMOR := Color("c4dcec")
const ACCENT := Color("ff914d")

var world: Node

var root_control: Control
var status_panel: PanelContainer
var objective_panel: PanelContainer
var action_panel: PanelContainer
var feedback_panel: PanelContainer
var speed_panel: PanelContainer
var help_panel: PanelContainer

var money_value: Label
var wanted_value: Label
var weapon_value: Label
var ammo_value: Label
var health_value: Label
var armor_value: Label
var health_bar: ProgressBar
var armor_bar: ProgressBar
var thermal_value: Label
var objective_value: Label
var action_value: Label
var feedback_value: Label
var help_value: Label
var speed_value: Label

var _session: Node
var _driving: Node
var _controls: Node
var _legacy_stats: Label
var _legacy_objective: Label
var _legacy_prompt: Label
var _legacy_notice: Label
var _legacy_thermal: Label
var _legacy_driving_prompt: Label
var _legacy_help: Label
var _legacy_speed: Label
var _bound := false
var _status_clock := 0.0

func _ready() -> void:
	name = "GameplayHUD"
	layer = 50
	process_mode = Node.PROCESS_MODE_ALWAYS
	process_priority = 1000
	_build_interface()
	get_viewport().size_changed.connect(_layout)
	_layout()

func _build_interface() -> void:
	root_control = Control.new()
	root_control.name = "GameplayHUDRoot"
	root_control.mouse_filter = Control.MOUSE_FILTER_IGNORE
	root_control.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	add_child(root_control)

	status_panel = _make_panel("StatusPanel", false)
	var status := _panel_column(status_panel, 10, 8)
	var top := HBoxContainer.new()
	top.add_theme_constant_override("separation", 12)
	status.add_child(top)
	money_value = _label("Money", 24, MONEY, true)
	money_value.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	top.add_child(money_value)
	wanted_value = _label("Wanted", 17, WANTED, true)
	wanted_value.horizontal_alignment = HORIZONTAL_ALIGNMENT_RIGHT
	top.add_child(wanted_value)

	var weapon_row := HBoxContainer.new()
	weapon_row.add_theme_constant_override("separation", 10)
	status.add_child(weapon_row)
	weapon_value = _label("Weapon", 19, TEXT, true)
	weapon_value.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	weapon_row.add_child(weapon_value)
	ammo_value = _label("Ammo", 17, TEXT, true)
	ammo_value.horizontal_alignment = HORIZONTAL_ALIGNMENT_RIGHT
	weapon_row.add_child(ammo_value)

	var health := _vital_row("VIDA", HEALTH)
	status.add_child(health.row)
	health_bar = health.bar
	health_value = health.value
	var armor := _vital_row("COLETE", ARMOR)
	status.add_child(armor.row)
	armor_bar = armor.bar
	armor_value = armor.value
	thermal_value = _label("Thermal", 14, MUTED, false)
	thermal_value.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	status.add_child(thermal_value)

	objective_panel = _make_panel("ObjectivePanel", true)
	var objective_column := _panel_column(objective_panel, 10, 4)
	var objective_header := _label("ObjectiveHeader", 11, ACCENT, true)
	objective_header.text = "OBJETIVO"
	objective_header.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	objective_column.add_child(objective_header)
	objective_value = _label("Objective", 18, TEXT, true)
	objective_value.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	objective_value.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	objective_value.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	objective_column.add_child(objective_value)

	action_panel = _make_panel("ActionPanel", true)
	var action_margin := _panel_margin(action_panel, 13, 8)
	action_value = _label("NearbyAction", 20, TEXT, true)
	action_value.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	action_value.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	action_margin.add_child(action_value)

	feedback_panel = _make_panel("FeedbackPanel", false)
	var feedback_margin := _panel_margin(feedback_panel, 13, 7)
	feedback_value = _label("TemporaryMessage", 17, Color("fff0cf"), true)
	feedback_value.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	feedback_value.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	feedback_margin.add_child(feedback_value)

	speed_panel = _make_panel("SpeedPanel", false)
	var speed_margin := _panel_margin(speed_panel, 12, 6)
	speed_value = _label("VehicleSpeed", 24, TEXT, true)
	speed_value.horizontal_alignment = HORIZONTAL_ALIGNMENT_RIGHT
	speed_margin.add_child(speed_value)

	help_panel = _make_panel("ContextHelpPanel", false)
	var help_margin := _panel_margin(help_panel, 9, 2)
	help_value = _label("ContextHelp", 14, Color("dbe1e5"), true)
	help_value.name = "ContextHelp"
	help_value.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	help_value.text_overrun_behavior = TextServer.OVERRUN_TRIM_ELLIPSIS
	help_value.mouse_filter = Control.MOUSE_FILTER_IGNORE
	help_margin.add_child(help_value)

func _make_panel(node_name: String, accented: bool) -> PanelContainer:
	var panel := PanelContainer.new()
	panel.name = node_name
	panel.mouse_filter = Control.MOUSE_FILTER_IGNORE
	panel.add_theme_stylebox_override("panel", _panel_style(accented))
	root_control.add_child(panel)
	return panel

func _panel_style(accented: bool) -> StyleBoxFlat:
	var style := StyleBoxFlat.new()
	style.bg_color = Color(0.025, 0.035, 0.045, 0.84)
	style.border_color = Color(1.0, 0.57, 0.30, 0.78) if accented else Color(0.30, 0.37, 0.42, 0.78)
	style.set_border_width_all(1)
	if accented: style.border_width_left = 3
	style.set_corner_radius_all(7)
	style.shadow_color = Color(0, 0, 0, 0.32)
	style.shadow_size = 5
	return style

func _panel_margin(panel: PanelContainer, horizontal: int, vertical: int) -> MarginContainer:
	var margin := MarginContainer.new()
	margin.add_theme_constant_override("margin_left", horizontal)
	margin.add_theme_constant_override("margin_right", horizontal)
	margin.add_theme_constant_override("margin_top", vertical)
	margin.add_theme_constant_override("margin_bottom", vertical)
	panel.add_child(margin)
	return margin

func _panel_column(panel: PanelContainer, horizontal: int, separation: int) -> VBoxContainer:
	var margin := _panel_margin(panel, horizontal, 9)
	var column := VBoxContainer.new()
	column.add_theme_constant_override("separation", separation)
	margin.add_child(column)
	return column

func _label(node_name: String, size_value: int, color: Color, strong: bool) -> Label:
	var label := Label.new()
	label.name = node_name
	label.mouse_filter = Control.MOUSE_FILTER_IGNORE
	label.add_theme_font_override("font", FONT_SEMIBOLD if strong else FONT_REGULAR)
	label.add_theme_font_size_override("font_size", size_value)
	label.set_meta("base_font_size", size_value)
	label.add_theme_color_override("font_color", color)
	label.add_theme_color_override("font_outline_color", Color(0, 0, 0, 0.98))
	label.add_theme_constant_override("outline_size", 3 if strong else 2)
	return label

func _vital_row(title: String, color: Color) -> Dictionary:
	var row := HBoxContainer.new()
	row.name = title.capitalize() + "Row"
	row.add_theme_constant_override("separation", 8)
	var caption := _label(title + "Caption", 13, MUTED, true)
	caption.text = title
	caption.custom_minimum_size.x = 48
	row.add_child(caption)
	var bar := ProgressBar.new()
	bar.name = title.capitalize() + "Bar"
	bar.min_value = 0
	bar.max_value = 100
	bar.show_percentage = false
	bar.custom_minimum_size = Vector2(160, 9)
	bar.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	bar.size_flags_vertical = Control.SIZE_SHRINK_CENTER
	var background := StyleBoxFlat.new()
	background.bg_color = Color("090d10")
	background.set_corner_radius_all(3)
	var fill := StyleBoxFlat.new()
	fill.bg_color = color
	fill.set_corner_radius_all(3)
	bar.add_theme_stylebox_override("background", background)
	bar.add_theme_stylebox_override("fill", fill)
	row.add_child(bar)
	var value := _label(title + "Value", 14, TEXT, true)
	value.custom_minimum_size.x = 28
	value.horizontal_alignment = HORIZONTAL_ALIGNMENT_RIGHT
	row.add_child(value)
	return {"row": row, "bar": bar, "value": value}

func _process(delta: float) -> void:
	if not _bound:
		_try_bind()
		if not _bound: return
	_hide_legacy_sources()
	_sync_channels()
	_status_clock += delta
	if _status_clock >= 0.08:
		_status_clock = 0.0
		_sync_status()

func _try_bind() -> void:
	if world == null or not is_instance_valid(world): return
	_session = world.get("session")
	_driving = world.get("driving")
	_controls = get_node_or_null("/root/GameInput")
	if not is_instance_valid(_session) or not is_instance_valid(_driving): return
	_legacy_stats = _session.get("stats")
	_legacy_objective = _session.get("objective")
	_legacy_prompt = _session.get("prompt")
	_legacy_notice = _session.get("notice")
	_legacy_thermal = _session.get("thermal_status")
	_legacy_driving_prompt = _driving.get("prompt")
	_legacy_speed = _driving.get("speed_label")
	_legacy_help = get_node_or_null("Help")
	if not is_instance_valid(_legacy_stats) or not is_instance_valid(_legacy_objective) or not is_instance_valid(_legacy_prompt) or not is_instance_valid(_legacy_notice): return
	_bound = true
	var title := get_node_or_null("Title")
	if is_instance_valid(title): title.hide()
	_layout()
	_sync_status()

func _hide_legacy_sources() -> void:
	for source in [_legacy_stats, _legacy_objective, _legacy_prompt, _legacy_notice, _legacy_thermal, _legacy_driving_prompt, _legacy_help, _legacy_speed]:
		if is_instance_valid(source): source.hide()

func _sync_channels() -> void:
	var modal: bool = bool(_session.get("modal"))
	var input_locked: bool = bool(world.player.get("input_locked")) if is_instance_valid(world.player) else true
	status_panel.visible = not modal
	objective_panel.visible = not modal
	help_panel.visible = not modal and not input_locked

	var objective_text := str(_legacy_objective.text).strip_edges()
	if objective_text == "J  Missões · M  Mapa":
		objective_text = "%s  Missões · %s  Mapa" % [_hint("journal", "J"), _hint("world_map", "M")]
	objective_value.text = objective_text
	objective_panel.visible = objective_panel.visible and not objective_text.is_empty()

	var actions: PackedStringArray = []
	var session_action := _format_action(str(_legacy_prompt.text), "interact", "E")
	if not session_action.is_empty(): actions.append(session_action)
	if float(_driving.get("status_time")) <= 0.0:
		var driving_action := _format_driving_action(str(_legacy_driving_prompt.text))
		if not driving_action.is_empty() and driving_action not in actions: actions.append(driving_action)
	action_value.text = "    ·    ".join(actions)
	action_panel.visible = not modal and not input_locked and not actions.is_empty()

	var feedback := ""
	if float(_session.get("notice_time")) > 0.0:
		feedback = str(_legacy_notice.text).strip_edges()
	elif float(_driving.get("status_time")) > 0.0:
		feedback = str(_driving.get("status")).strip_edges()
	feedback_value.text = feedback
	feedback_panel.visible = not feedback.is_empty()

	var occupied: bool = bool(_driving.get("occupied"))
	speed_value.text = str(_legacy_speed.text).strip_edges()
	speed_panel.visible = occupied and not modal and not speed_value.text.is_empty()
	help_value.text = _context_help(occupied)
	_layout_modal()

func _sync_status() -> void:
	if not is_instance_valid(_session) or _session.get("state") == null: return
	var state: Variant = _session.get("state")
	var gameplay: Variant = world.get("gameplay")
	if gameplay == null: return
	money_value.text = "R$ %d" % int(state.economy.balance)
	var stars: int = clampi(int(gameplay.stars), 0, 6)
	wanted_value.text = "SEM PROCURA" if stars == 0 else "PROCURADO  %s%s" % ["★".repeat(stars), "☆".repeat(6-stars)]
	wanted_value.add_theme_color_override("font_color", MUTED if stars == 0 else WANTED)
	var weapon_id: String = str(state.equipped_weapon)
	weapon_value.text = str(WEAPONS.WEAPONS.get(weapon_id, {}).get("label", weapon_id if not weapon_id.is_empty() else "Mãos livres"))
	var ammo: Dictionary = state.get_ammo(weapon_id)
	var magazine: int = int(ammo.get("magazine", -1))
	ammo_value.text = "MUNIÇÃO  —" if magazine < 0 else "PENTE %d  ·  RESERVA %d" % [magazine, maxi(0, int(ammo.get("reserve", 0)))]
	var health: int = clampi(roundi(float(gameplay.health)), 0, 100)
	var armor: int = clampi(roundi(float(gameplay.armor)), 0, 100)
	health_bar.value = health
	armor_bar.value = armor
	health_value.text = str(health)
	armor_value.text = str(armor)
	var thermal_text := ""
	if _session.get("cold") != null:
		var thermal: Dictionary = _session.cold.status()
		if bool(thermal.get("visible", false)):
			thermal_text = "Calor corporal %d%% · %s" % [roundi(float(thermal.get("temperature", 0))), str(thermal.get("text", ""))]
	thermal_value.text = thermal_text
	thermal_value.visible = not thermal_text.is_empty()

func _format_action(text: String, action: String, legacy: String) -> String:
	text = text.strip_edges()
	if text.is_empty(): return ""
	for prefix in [legacy + "  ", legacy + " ", "[" + legacy + "] "]:
		if text.begins_with(prefix):
			text = text.trim_prefix(prefix).strip_edges()
			break
	return "%s  %s" % [_hint(action, legacy), text]

func _format_driving_action(text: String) -> String:
	text = text.strip_edges()
	if text.is_empty(): return ""
	if "Sair do carro" in text: return _format_action(text, "exit_vehicle", "F")
	if "Entrar no carro" in text: return _format_action(text, "vehicle_interact", "F")
	return text

func _hint(action: String, fallback: String) -> String:
	if is_instance_valid(_controls) and _controls.has_method("hint"):
		var result := str(_controls.hint(action))
		if not result.is_empty() and result != "—": return result
	return fallback

func _context_help(occupied: bool) -> String:
	if occupied:
		if is_instance_valid(_controls) and bool(_controls.get("using_gamepad")):
			return "%s acelerar  ·  %s frear/ré  ·  %s freio de mão  ·  %s sair  ·  %s pausa" % [_hint("accelerate", "RT"), _hint("brake", "LT"), _hint("handbrake", "A"), _hint("exit_vehicle", "Y"), _hint("pause_game", "Menu")]
		return "%s acelerar  ·  %s ré  ·  %s/%s virar  ·  %s freio  ·  %s sair  ·  %s pausa" % [_primary_hint("move_up", "W"), _primary_hint("move_down", "S"), _primary_hint("move_left", "A"), _primary_hint("move_right", "D"), _primary_hint("handbrake", "Espaço"), _primary_hint("exit_vehicle", "F"), _primary_hint("pause_game", "Esc")]
	if is_instance_valid(_controls) and bool(_controls.get("using_gamepad")):
		return "%s mover  ·  %s correr  ·  %s interagir  ·  %s veículo  ·  %s pausa" % [_hint("move_up", "Analógico E"), _hint("sprint", "L3"), _hint("interact", "X"), _hint("vehicle_interact", "Y"), _hint("pause_game", "Menu")]
	return "%s/%s/%s/%s mover  ·  %s correr  ·  %s interagir  ·  %s veículo  ·  %s pausa" % [_primary_hint("move_up", "W"), _primary_hint("move_left", "A"), _primary_hint("move_down", "S"), _primary_hint("move_right", "D"), _primary_hint("sprint", "Shift"), _primary_hint("interact", "E"), _primary_hint("vehicle_interact", "F"), _primary_hint("pause_game", "Esc")]

func _primary_hint(action: String, fallback: String) -> String:
	var result := _hint(action, fallback)
	return result.get_slice(" / ", 0) if " / " in result else result

func _layout() -> void:
	if not is_instance_valid(root_control): return
	var dimensions := get_viewport().get_visible_rect().size
	var scale := clampf(minf(dimensions.x / 1280.0, dimensions.y / 720.0), 0.82, 1.18)
	var margin := 24.0 * scale
	var status_width := 350.0 * scale
	var status_height := 164.0 * scale
	_place(status_panel, Vector2(dimensions.x-margin-status_width, margin), Vector2(status_width, status_height))
	var narrow := dimensions.x < 1080.0
	var objective_width := minf(620.0 * scale, dimensions.x - margin * 2.0)
	var objective_top := 190.0 * scale if narrow else margin
	var objective_left := (dimensions.x-objective_width)*0.5
	if not narrow:
		var available_width := dimensions.x-status_width-margin*3.0
		objective_width = minf(objective_width, available_width)
		objective_left = margin+(available_width-objective_width)*0.5
	_place(objective_panel, Vector2(objective_left, objective_top), Vector2(objective_width, 66.0*scale))
	var feedback_top := objective_top + 78.0 * scale
	_place(feedback_panel, Vector2(objective_left, feedback_top), Vector2(objective_width, 48.0*scale))
	var action_width := minf(620.0 * scale, dimensions.x - margin * 2.0)
	_place(action_panel, Vector2((dimensions.x-action_width)*0.5, dimensions.y-104.0*scale), Vector2(action_width, 54.0*scale))
	_place(speed_panel, Vector2(dimensions.x-margin-190.0*scale, dimensions.y-104.0*scale), Vector2(190.0*scale, 54.0*scale))
	_place(help_panel, Vector2(margin, dimensions.y-40.0*scale), Vector2(maxf(160.0, dimensions.x-margin*2.0), 30.0*scale))
	for label in [money_value, wanted_value, weapon_value, ammo_value, health_value, armor_value, thermal_value, objective_value, action_value, feedback_value, help_value, speed_value]:
		if is_instance_valid(label): label.add_theme_font_size_override("font_size", maxi(11, roundi(float(label.get_meta("base_font_size", 14))*scale)))
	_layout_modal()

func _place(control: Control, point: Vector2, extent: Vector2) -> void:
	if not is_instance_valid(control): return
	control.set_anchors_preset(Control.PRESET_TOP_LEFT)
	control.position = point.round()
	control.size = extent.round()
	control.custom_minimum_size = extent.round()

func _layout_modal() -> void:
	if not _bound or not is_instance_valid(_session.get("panel")): return
	var modal_panel: Control = _session.panel
	var dimensions := get_viewport().get_visible_rect().size
	var extent := Vector2(minf(620.0, dimensions.x-48.0), minf(440.0, dimensions.y-48.0))
	var preferred: Variant = modal_panel.get_meta("modal_preferred_size") if modal_panel.has_meta("modal_preferred_size") else null
	if preferred is Vector2:
		extent=Vector2(minf(preferred.x,dimensions.x-48.0),minf(preferred.y,dimensions.y-48.0))
		modal_panel.custom_minimum_size=extent.round()
		modal_panel.size=extent.round()
	modal_panel.position = ((dimensions-extent)*0.5).round()
	modal_panel.custom_minimum_size = extent.round()
